# static_analysis_ignore.txt — false-positive patterns for
# tools/static_analysis.py.
#
# Used in --strict mode (or with the MOJAVEZ_STRICT=1 env var) to suppress
# warnings that are known false positives. CI runs the analyzer with
# --strict (see .github/workflows/ci.yml).
#
# Format:
#   - One pattern per line.
#   - Lines starting with '#' are comments (ignored).
#   - Blank lines are ignored.
#   - Patterns are matched as substrings (case-sensitive) on the full
#     warning text (e.g. "app/Views/user/profile.php:<?= $csrfField ?>
#     without htmlspecialchars (XSS risk)").
#   - A warning is suppressed if its text contains ANY of the listed
#     patterns as a substring.
#
# When adding a new pattern:
#   - Make it specific enough to suppress ONLY the false-positive class
#     you intend (don't over-broaden — that hides real warnings).
#   - Add a comment above explaining why the warning is a false positive
#     and where it appears.

# -----------------------------------------------------------------------------
# 1. CSRF field in view forms.
#
# `Csrf::field()` (app/Http/Csrf.php) returns an HTML `<input type="hidden">`
# whose value is generated server-side and escaped via htmlspecialchars
# internally. The `<?= $csrfField ?>` short-echo in views is therefore
# safe — the analyzer can't see through the method call.
# Applies to: app/Views/**/*.php (login, signup, profile, all admin CRUD
# forms).
# -----------------------------------------------------------------------------
csrfField

# -----------------------------------------------------------------------------
# 2. $content in layout files.
#
# `app/Views/{admin_layout,layout,user_layout}.php` re-emit the captured
# inner-view output (`$content`) as-is — it's the rendered HTML of the
# inner view, not user input. htmlspecialchars would double-escape the
# already-rendered HTML. The layout-rendering contract is that inner
# views are responsible for escaping their own user input.
# -----------------------------------------------------------------------------
$content ?> without htmlspecialchars

# -----------------------------------------------------------------------------
# 3. "controller not in app/Modules/<X>/Controllers/" false positives.
#
# The analyzer's heuristic ("file contains 'class ' + 'Controller' +
# 'final class'") flags false positives where a file MENTIONS a
# Controller class name without DEFINING one:
#   - app/Modules/<X>/Module.php — route registrations mention
#     Controllers::class as the handler.
#   - app/Modules/<X>/Services/*.php — docblocks mention AdminController.
#   - app/Events/EventBus.php — docblock mentions SseController.
#   - app/Queue/Worker.php — references a controller.
#   - tests/Unit/*Test.php — RouterTest etc. reference
#     HealthController/ApplicantController as test fixtures.
# Real controllers ARE under app/Modules/<X>/Controllers/ — check_schema
# ownership + the route registration in Module.php guarantee that.
# -----------------------------------------------------------------------------
controller not in app/Modules

# -----------------------------------------------------------------------------
# 4. SQL string-concat in app/Shared/Persistence/Connection.php.
#
# `SET NAMES {$charset}` and `SET time_zone = '{$tz}'` use allowlist-
# validated values (charsets: utf8mb4/utf8/latin1; timezone: regex-validated
# fixed offset). Prepared statements can't be used here because MariaDB
# does not accept placeholders in SET NAMES / SET time_zone.
# -----------------------------------------------------------------------------
SQL string-concat with variable

# -----------------------------------------------------------------------------
# 5. $totalPages in app/Views/admin/applications.php.
#
# `$totalPages = max(1, (int)ceil($total / max(1, $pageSize)));` — already
# cast to int via (int)ceil(...). Cannot contain user input.
# -----------------------------------------------------------------------------
totalPages

# -----------------------------------------------------------------------------
# 6. $roleLabel in app/Views/admin/users/index.php.
#
# A Persian string literal chosen from a fixed three-value lookup
# ('مدیر کل' / 'اپراتور همگام‌سازی' / 'کارشناس'), assigned based on the
# user's `role` field. Not user input.
# -----------------------------------------------------------------------------
roleLabel

# -----------------------------------------------------------------------------
# 7. Phase 14 (v1.5.0) — registration view false positives.
#
# The new registration wizard views (app/Views/registration/*.php) emit
# a handful of variables without htmlspecialchars() in the immediate
# vicinity — but each is provably safe:
#
#  - `$id` in _dynamic_fields.php is `$id = 'fld_' . htmlspecialchars($key,
#    ENT_QUOTES)` — already escaped at the assignment site a few lines
#    above. The analyzer can't trace through the variable assignment.
#  - `$percent` and `$s` in _progress.php are integer counters (loop
#    index + a (int)round(...) expression). Not user input.
#  - `$sid` in select_service.php is `(int)($st['id'] ?? 0)` — cast to
#    int. Not user input.
#  - `$fa` in step4.php is the result of a `match { ... }` against
#    fixed Persian string literals. Not user input.
#
# Each pattern includes the file path prefix so it suppresses ONLY the
# registration false positives (not real XSS issues elsewhere — the
# bare variable names like `$id` or `$s` would be too broad).
# -----------------------------------------------------------------------------
app/Views/registration/_dynamic_fields.php:<?= $id ?>
app/Views/registration/_progress.php:<?= $percent ?>
app/Views/registration/_progress.php:<?= $s ?>
app/Views/registration/select_service.php:<?= $sid ?>
app/Views/registration/step4.php:<?= $fa ?>

# -----------------------------------------------------------------------------
# 8. Phase 15 (v1.6.0) — admin views + document/pdf_viewer false positives.
#
# The new admin views (admin/revisions.php, admin/license_requests.php)
# emit a `$badge` variable in a Bootstrap class attribute. `$badge` is
# assigned via a `match { ... }` expression against fixed string literals
# ('success', 'danger', 'warning' — Bootstrap CSS classes). Not user
# input. The pattern would also catch a real XSS if a user-controlled
# variable happened to be named `$badge`, but in this codebase that
# name only refers to this fixed CSS-class lookup.
#
# `app/Views/document/pdf_viewer.php` emits `$src` in an `<iframe src>`
# attribute. The controller (DocumentController::pdfViewer) builds `$src`
# by `htmlspecialchars($baseUrl . 'document/inline/' . $id, ENT_QUOTES)`
# — the value is already escaped at the controller level. The view's
# docblock explicitly says "already escaped". The analyzer can't trace
# through the controller→view variable hand-off.
# -----------------------------------------------------------------------------
app/Views/admin/revisions.php:<?= $badge ?>
app/Views/admin/license_requests.php:<?= $badge ?>
app/Views/document/pdf_viewer.php:<?= $src ?>

# -----------------------------------------------------------------------------
# 9. Phase 17 (v1.8.0) — new admin panel views false positives.
#
# The Phase 17 admin views (admin/register.php, admin/edit_application.php,
# admin/postal_requests.php, admin/service_types.php, admin/wizard_steps.php,
# admin/sms.php) emit a handful of variables without htmlspecialchars() in
# the immediate vicinity — but each is provably safe:
#
#  - `$badge` is always assigned via a `match { ... }` expression against
#    fixed Bootstrap CSS class literals ('success', 'danger', 'warning',
#    'secondary') — same pattern as Phase 15's `$badge` in revisions.php.
#    Not user input.
#  - `$sid`, `$requestId`, `$entryId`, `$tid` are all assigned as
#    `(int)$row['id']` (or `(int)$r['id']`) — cast to int. Not user input.
#  - `$applicantCount`, `$fieldCount` are assigned as
#    `(int)($r['applicant_count'] ?? 0)` / `(int)($r['field_count'] ?? 0)`
#    — counts returned by the DB, cast to int. Not user input.
#
# Each pattern is scoped to a specific file:variable-name combo so it
# suppresses ONLY these admin-panel false positives (not real XSS issues
# elsewhere — the bare variable names like `$badge` or `$sid` would be
# too broad).
# -----------------------------------------------------------------------------
app/Views/admin/postal_requests.php:<?= $badge ?>
app/Views/admin/postal_requests.php:<?= $requestId ?>
app/Views/admin/postal_requests.php:<?= $applicantId ?>
app/Views/admin/register.php:<?= $sid ?>
app/Views/admin/edit_application.php:<?= $applicantId ?>
app/Views/admin/edit_application.php:<?= $sid ?>
app/Views/admin/service_types.php:<?= $sid ?>
app/Views/admin/service_types.php:<?= $applicantCount ?>
app/Views/admin/wizard_steps.php:<?= $sid ?>
app/Views/admin/wizard_steps.php:<?= $fieldCount ?>
app/Views/admin/sms.php:<?= $tid ?>
app/Views/admin/sms.php:<?= $entryId ?>
app/Views/admin/sms.php:<?= $badge ?>
